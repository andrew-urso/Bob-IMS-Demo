## Note for bob
## This is an example DDL to create a simple DATABASE
## IMS DDL has various specific parameters and restrictions
## that other forms of DDL do not have
CREATE DATABASE DI23PART ACCESS HISAM
    VERSION '01/11/1814.19'
    PASSWDNO
    DATXEXITNO
    DATA CAPTURE NONE CCSID 'Cp1047';

CREATE TABLESPACE DI23PART IN DI23PART
    RECORD(678,678)
    SIZE PRIMARY 2048 SIZE SECONDARY 2048
    OVERFLOW(DI23PARO);

CREATE TABLE PARTROOT (
    PARTKEY CHAR(17)
        START 1
        TYPE C
        INTERNALNAME PARTKEY
        PRIMARY KEY
        INTERNAL TYPECONVERTER CHAR
        CCSID 'Cp1047',
    PARTNAME CHAR(12)
        START 18
        TYPE C
        INTERNALNAME PARTNAME
        INTERNAL TYPECONVERTER CHAR
        CCSID 'Cp1047',
    PARTDESC CHAR(21)
        START 22
        TYPE C
        INTERNALNAME PARTDESC
        INTERNAL TYPECONVERTER CHAR
        CCSID 'Cp1047'
    ) IN DATABASE DI23PART
    INTERNALNAME PARTROOT
    MAXBYTES 50
    INSERT LOGICAL DELETE LOGICAL REPLACE LOGICAL
    AMBIGUOUS INSERT LAST
    CCSID 'Cp1047';

COMMIT DDL;
